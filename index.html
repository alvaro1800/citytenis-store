document.addEventListener('DOMContentLoaded', () => { const carrusel = document.querySelector('.carrusel'); const puntosContainer = document.querySelector('.puntos'); // SOLO EJECUTA SI HAY CARRUSEL EN LA PÁGINA if (carrusel && puntosContainer) { const imagenes = carrusel.querySelectorAll('img'); let indiceActual = 0; // Crear puntitos imagenes.forEach((_, i) => { const punto = document.createElement('button'); punto.classList.add('punto'); if (i === 0) punto.classList.add('activo'); punto.addEventListener('click', () => { carrusel.scrollTo({ left: carrusel.offsetWidth * i, behavior: 'smooth' }); }); puntosContainer.appendChild(punto); }); const puntos = puntosContainer.querySelectorAll('.punto'); // Detectar scroll carrusel.addEventListener('scroll', () => { indiceActual = Math.round(carrusel.scrollLeft / carrusel.offsetWidth); puntos.forEach((p, i) => p.classList.toggle('activo', i === indiceActual)); }); } });
